    47→// ─── Password Strength ───
    48→
    49→function getPasswordStrength(password: string) {
    50→  let score = 0;
    51→  if (password.length >= 6) score++;
    52→  if (password.length >= 8) score++;
    53→  if (/[A-Z]/.test(password)) score++;
    54→  if (/[0-9]/.test(password)) score++;
    55→  if (/[^A-Za-z0-9]/.test(password)) score++;
    56→  if (score <= 1) return { score, labelKey: 'weak', color: 'bg-red-500' };
    57→  if (score <= 2) return { score, labelKey: 'fair', color: 'bg-yellow-500' };
    58→  if (score <= 3) return { score, labelKey: 'good', color: 'bg-blue-500' };
    59→  return { score, labelKey: 'strong', color: 'bg-teal-500' };
    60→}

    64→function StepIndicator({ currentStep, showMigration }: { currentStep: WizardStep; showMigration: boolean }) {
    65→  const { t } = useTranslations('setup');
    66→  const steps = [
    67→    ...(showMigration ? [{ key: 'db-migration' as const, label: t('dbSetup'), num: 0 }] : []),
    68→    { key: 'admin-account' as const, label: t('adminAccount'), num: showMigration ? 2 : 1 },
    69→    { key: 'institution-info' as const, label: t('institutionData'), num: showMigration ? 3 : 2 },
    70→    { key: 'complete' as const, label: t('done'), num: showMigration ? 4 : 3 },
    71→  ];
    72→
    73→  const currentIndex = steps.findIndex((s) => s.key === currentStep);
    74→
    75→  return (
    76→    <div className="flex items-center justify-center gap-2 mb-8">
    77→      {steps.map((step, idx) => {
    78→        const isActive = idx === currentIndex;
    79→        const isDone = idx < currentIndex;
    80→        return (
    81→          <div key={step.key} className="flex items-center gap-2">
    82→            <div className="flex items-center gap-1.5">
    83→              <div
    84→                className={`flex h-8 w-8 items-center justify-center rounded-full text-sm font-bold transition-all duration-300 ${
    85→                  isDone
    86→                    ? 'bg-sky-600 text-white'
    87→                    : isActive
    88→                    ? 'bg-sky-700 text-white shadow-lg shadow-sky-500/30'
    89→                    : 'bg-white/20 text-white/60'
    90→                }`}
    91→              >
    92→                {isDone ? <CheckCircle2 className="h-4 w-4" /> : step.num}
    93→              </div>
    94→              <span
    95→                className={`text-xs font-medium transition-colors ${
    96→                  isActive ? 'text-white' : isDone ? 'text-sky-200' : 'text-white/50'
    97→                }`}
    98→              >
    99→                {step.label}
   100→              </span>
   101→            </div>
   102→            {idx < steps.length - 1 && (
   103→              <div
   104→                className={`h-0.5 w-8 rounded-full transition-colors ${

   107→              />
   108→            )}

   532→-- 3. Setup initialization RPC function
   533→CREATE OR REPLACE FUNCTION setup_initialize_system(
   534→  p_name TEXT, p_name_en TEXT DEFAULT NULL,
   535→  p_type TEXT DEFAULT 'center', p_logo_url TEXT DEFAULT NULL,

   542→)
   543→RETURNS JSON LANGUAGE plpgsql SECURITY DEFINER AS $$
   544→DECLARE v_id UUID; v_existing_id UUID;
   545→BEGIN
   546→  SELECT id INTO v_existing_id FROM institution_settings LIMIT 1;
   547→  IF v_existing_id IS NOT NULL THEN
   548→    UPDATE institution_settings SET name=p_name, name_en=p_name_en, type=p_type,
   549→      logo_url=p_logo_url, tagline=p_tagline, country=p_country, city=p_city, address=p_address,
   550→      phone=p_phone, email=p_email, website=p_website, timezone=p_timezone,
   551→      academic_year=p_academic_year, description=p_description
   552→    WHERE id=v_existing_id;
   553→    RETURN json_build_object('action','updated','id',v_existing_id);
