    47→    47→// ─── Password Strength ───

    49→    49→function getPasswordStrength(password: string) {
    50→    50→  let score = 0;
    51→    51→  if (password.length >= 6) score++;
    52→    52→  if (password.length >= 8) score++;
    53→    53→  if (/[A-Z]/.test(password)) score++;
    54→    54→  if (/[0-9]/.test(password)) score++;
    55→    55→  if (/[^A-Za-z0-9]/.test(password)) score++;
    56→    56→  if (score <= 1) return { score, labelKey: 'weak', color: 'bg-red-500' };
    57→    57→  if (score <= 2) return { score, labelKey: 'fair', color: 'bg-yellow-500' };
    58→    58→  if (score <= 3) return { score, labelKey: 'good', color: 'bg-blue-500' };
    59→    59→  return { score, labelKey: 'strong', color: 'bg-teal-500' };
    60→    60→}
